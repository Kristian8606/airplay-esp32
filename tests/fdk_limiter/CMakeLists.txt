cmake_minimum_required(VERSION 3.16)
project(fdk_limiter_host C CXX)
set(CMAKE_CXX_STANDARD 11)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(FDK "${ROOT}/components/fdk_aac/fdk-aac")
set(SRC)
set(INC)
foreach(lib libAACdec libArithCoding libDRCdec libFDK libMpegTPDec libPCMutils libSACdec libSBRdec libSYS)
 file(GLOB files "${FDK}/${lib}/src/*.cpp")
 list(APPEND SRC ${files})
 list(APPEND INC "${FDK}/${lib}/include")
endforeach()
add_library(fdk STATIC ${SRC})
target_include_directories(fdk PUBLIC ${INC})
target_compile_options(fdk PRIVATE -O2 -w -fsanitize=bounds -fno-sanitize-recover=all)
add_executable(limiter_test limiter_test.cpp)
target_link_libraries(limiter_test PRIVATE fdk)
target_compile_options(limiter_test PRIVATE -fsanitize=bounds -fno-sanitize-recover=all)
target_link_options(limiter_test PRIVATE -fsanitize=bounds -fno-sanitize-recover=all)

add_executable(decode_probe decode_probe.cpp)
target_link_libraries(decode_probe PRIVATE fdk)
target_link_options(decode_probe PRIVATE -fsanitize=bounds -fno-sanitize-recover=all)

foreach(mode on off)
 add_executable(wrapper_${mode} wrapper_test.cpp "${ROOT}/main/audio/aac_decoder.c" "${ROOT}/main/audio/pcm_fifo.c")
 target_compile_definitions(wrapper_${mode} PRIVATE PCM_FIFO_HOST_TEST=1)
 target_include_directories(wrapper_${mode} PRIVATE "${ROOT}/main" "${ROOT}/main/audio" "${ROOT}/tests/homekit/stubs")
 target_link_libraries(wrapper_${mode} PRIVATE fdk)
 target_link_options(wrapper_${mode} PRIVATE -fsanitize=bounds -fno-sanitize-recover=all)
 if(mode STREQUAL "on")
  target_compile_definitions(wrapper_${mode} PRIVATE CONFIG_AIRPLAY_AAC_FDK_LIMITER=1 LIMITER_EXPECTED=1)
 else()
  target_compile_definitions(wrapper_${mode} PRIVATE CONFIG_AIRPLAY_AAC_FDK_LIMITER=0 LIMITER_EXPECTED=0)
 endif()
endforeach()

add_executable(transport_test transport_test.c)
target_include_directories(transport_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/stubs" "${ROOT}/tests/homekit/stubs" "${ROOT}/main" "${ROOT}/main/audio" "${ROOT}/components/helpers/include")
target_compile_definitions(transport_test PRIVATE CONFIG_AIRPLAY_AAC_FDK_LIMITER=1 ESP_ERR_TIMEOUT=0x107)
target_compile_options(transport_test PRIVATE -fsanitize=undefined -fno-sanitize-recover=all)
target_link_options(transport_test PRIVATE -fsanitize=undefined -fno-sanitize-recover=all)
